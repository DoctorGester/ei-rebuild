# Lightning bolts (lightning.cpp): three script bolts next to the hero with positive, zero and
# negative power (each power has its own branch layout and animation), drawn for a few seconds and
# deleted, then the lightning spell on a wolf (a bolt that follows its two units).
reject Script error
waitmenu
console thingamabob
newgame
waitfor 15 IsGreater(HP(GetLeader()),0)
console @BlockUnit(GetLeader(),0)

console @GSSetVar(0,"lx",GetX(GetLeader())) GSSetVar(0,"ly",GetY(GetLeader()))
console @CreateLightning(951,GSGetVar(0,"lx"),GSGetVar(0,"ly"),25,Add(GSGetVar(0,"lx"),3),GSGetVar(0,"ly"),0,5)
console @CreateLightning(952,Add(GSGetVar(0,"lx"),-3),GSGetVar(0,"ly"),25,GSGetVar(0,"lx"),Add(GSGetVar(0,"ly"),2),0,0)
console @CreateLightning(953,GSGetVar(0,"lx"),Add(GSGetVar(0,"ly"),-3),25,Add(GSGetVar(0,"lx"),2),GSGetVar(0,"ly"),0,-7)
wait 3000
console @DeleteLightning(951) DeleteLightning(952) DeleteLightning(953)
wait 500

console @AddUnitToServer("LtWolf","WolfTutor","LtWolf",Add(GetX(GetLeader()),5),GetY(GetLeader()),0)
waitfor 10 IsAlive(GetObjectByName("LtWolf"))
console @GSSetVar(0,"lw",HP(GetObjectByName("LtWolf")))
console @CastSpellUnit("lightning{e2;e2;e2;e2;e2}",GetX(GetLeader()),GetY(GetLeader()),GetObjectByName("LtWolf"))
waitfor 10 IsLess(HP(GetObjectByName("LtWolf")),GSGetVar(0,"lw"))
wait 1000
check IsAlive(GetLeader())
report
quit
