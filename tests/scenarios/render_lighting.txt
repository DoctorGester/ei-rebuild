# R16i: the world lit on the GPU (docs/notes/renderer.md section 13, item 9): frozen moments for the
# user's review and the frame checks. zone1's start (a slope beside the ruins), a fight of creatures
# around the hero, zone1 at night, the town bz1g by day and at night (its campfire). Frames only (image).
# no original: vanilla rebirth (the lighting is our renderer's)
render checkall
timeout 180
reject Script error
waitmenu
console thingamabob
newgame
waitfor 15 IsGreater(HP(GetLeader()),0)
waitfor 15 IsUnitBlocked(GetLeader())
console @BlockUnit(GetLeader(),0)
waitwindow 15 battle
waitpresent 10
move 400 20
frames 5
freeze
capture light_start
unfreeze

# Slopes: the hill right of the ruins and the ground left of them, three creatures on each.
console filter all on
console @GodMode(0,1)
console @GSSetVar(0,"hx",GetX(GetLeader()))
console @GSSetVar(0,"hy",GetY(GetLeader()))
console @AddUnitToServer("S1","WolfTutor","S1",Add(GSGetVar(0,"hx"),12),Add(GSGetVar(0,"hy"),1),0)
console @AddUnitToServer("S2","GoblinF1","S2",Add(GSGetVar(0,"hx"),14),Add(GSGetVar(0,"hy"),4),0)
console @AddUnitToServer("S3","BoarTutor","S3",Add(GSGetVar(0,"hx"),12),Add(GSGetVar(0,"hy"),7),0)
console @AddUnitToServer("S4","WolfTutor","S4",Sub(GSGetVar(0,"hx"),8),Add(GSGetVar(0,"hy"),8),0)
console @AddUnitToServer("S5","GoblinF1","S5",Sub(GSGetVar(0,"hx"),10),Add(GSGetVar(0,"hy"),7),0)
console @AddUnitToServer("S6","BoarTutor","S6",Sub(GSGetVar(0,"hx"),7),Add(GSGetVar(0,"hy"),10),0)
wait 3000
freeze
capture light_slope
unfreeze
console @RemoveUnitFromServer(GetObjectByName("S1"))
console @RemoveUnitFromServer(GetObjectByName("S2"))
console @RemoveUnitFromServer(GetObjectByName("S3"))
console @RemoveUnitFromServer(GetObjectByName("S4"))
console @RemoveUnitFromServer(GetObjectByName("S5"))
console @RemoveUnitFromServer(GetObjectByName("S6"))

# A fight: creatures around the hero, who cannot die.
console @AddUnitToServer("F1","WolfTutor","F1",Add(GSGetVar(0,"hx"),4),GSGetVar(0,"hy"),0)
console @AddUnitToServer("F2","WolfTutor","F2",Sub(GSGetVar(0,"hx"),4),GSGetVar(0,"hy"),0)
console @AddUnitToServer("F3","GoblinF1","F3",GSGetVar(0,"hx"),Add(GSGetVar(0,"hy"),4),0)
console @AddUnitToServer("F4","BoarTutor","F4",GSGetVar(0,"hx"),Sub(GSGetVar(0,"hy"),4),0)
console @AddUnitToServer("F5","SkeletonF2","F5",Add(GSGetVar(0,"hx"),5),Add(GSGetVar(0,"hy"),5),0)
console @AddUnitToServer("F6","GoblinF1","F6",Sub(GSGetVar(0,"hx"),5),Add(GSGetVar(0,"hy"),5),0)
waitfor 15 IsAlive(GetObjectByName("F6"))
console @SetPlayer(GetObjectByName("F1"),6)
console @SetPlayer(GetObjectByName("F2"),6)
console @SetPlayer(GetObjectByName("F3"),6)
console @SetPlayer(GetObjectByName("F4"),6)
console @SetPlayer(GetObjectByName("F5"),6)
console @SetPlayer(GetObjectByName("F6"),6)
console @Attack(GetObjectByName("F1"),GetLeader())
console @Attack(GetObjectByName("F2"),GetLeader())
console @Attack(GetObjectByName("F3"),GetLeader())
console @Attack(GetObjectByName("F4"),GetLeader())
console @Attack(GetObjectByName("F5"),GetLeader())
console @Attack(GetObjectByName("F6"),GetLeader())
wait 2500
freeze
capture light_fight
unfreeze
console @RemoveUnitFromServer(GetObjectByName("F1"))
console @RemoveUnitFromServer(GetObjectByName("F2"))
console @RemoveUnitFromServer(GetObjectByName("F3"))
console @RemoveUnitFromServer(GetObjectByName("F4"))
console @RemoveUnitFromServer(GetObjectByName("F5"))
console @RemoveUnitFromServer(GetObjectByName("F6"))
wait 500
check IsAlive(GetLeader())

# zone1 at night (the hour held at 1; the terrain relights over the next frames).
console @FixWorldTime(1)
wait 4000
freeze
capture light_night
unfreeze

# The town bz1g by day, then at night with its campfire.
zone bz1g 0
waitzone 20 bz1g
waitfor 15 IsGreater(HP(GetLeader()),0)
console @FixWorldTime(12)
wait 3000
move 400 20
frames 5
freeze
capture light_town
unfreeze
console @FixWorldTime(23)
wait 4000
frames 5
freeze
capture light_town_night
unfreeze
report
quit
