# ---- section movement
# Movement (src/game/ai/ai_motion.cpp, ai_movement.cpp, ai_path.cpp, sector_map.cpp): several units
# walk to points at the same time in the three movement modes (Walk, Run, Crawl), one goes to
# another unit (MoveToObject), two follow in a chain (Follow), and one walks to a point behind a
# standing ogre. Exercises the order header (AIPath Start/Reset/SetTarget), the trajectory (Build,
# Subdivide, LimitSpeeds, Sample every tick), the step speeds, the blocking-unit test and the
# sector map moves. Script signatures: Walk/Run/Crawl(unit), MoveToPoint(unit, x, y),
# MoveToObject(unit, target), Follow(unit, leader).
waitfor 15 IsGreater(HP(GetLeader()),0)
# The level intro keeps the hero blocked; unblock it instead of waiting.
console @BlockUnit(GetLeader(),0)
console @GSSetVar(0,"hx",GetX(GetLeader()))
console @GSSetVar(0,"hy",GetY(GetLeader()))

# Three wolves around the hero, without AI modes so only the script moves them.
# Mv1 starts where unit_ai.txt patrols (known walkable ground).
console @AddUnitToServer("Mv1","WolfTutor","Mv1",Add(GSGetVar(0,"hx"),4),GSGetVar(0,"hy"),0)
console @AddUnitToServer("Mv2","WolfTutor","Mv2",Sub(GSGetVar(0,"hx"),2),Add(GSGetVar(0,"hy"),2),0)
console @AddUnitToServer("Mv3","WolfTutor","Mv3",Add(GSGetVar(0,"hx"),2),Sub(GSGetVar(0,"hy"),2),0)
waitfor 10 IsAlive(GetObjectByName("Mv1"))
waitfor 10 IsAlive(GetObjectByName("Mv2"))
waitfor 10 IsAlive(GetObjectByName("Mv3"))
console @UMClear(GetObjectByName("Mv1"))
console @UMClear(GetObjectByName("Mv2"))
console @UMClear(GetObjectByName("Mv3"))

# The original game does not bring these units within 2.5 units of their targets in 20 s either
# (checked with --no-hooks), so the moves only run for a few seconds each; the per-tick motion
# code is compared side by side by its verify wrappers meanwhile.
# Three units walking to points at once: a wolf east, a wolf west, the hero two units south.
# (Walk/Run/Crawl before MoveToPoint left the units standing on the original game too; they are
# exercised at the end instead.)
console @MoveToPoint(GetObjectByName("Mv1"),Add(GSGetVar(0,"hx"),4),Add(GSGetVar(0,"hy"),4))
console @MoveToPoint(GetObjectByName("Mv2"),Sub(GSGetVar(0,"hx"),5),Add(GSGetVar(0,"hy"),2))
console @MoveToPoint(GetLeader(),GSGetVar(0,"hx"),Sub(GSGetVar(0,"hy"),2))
wait 4000

# MoveToObject: the third wolf goes to the first (5 east, 4 north of it).
console @MoveToObject(GetObjectByName("Mv3"),GetObjectByName("Mv1"))
wait 2000

# Follow in a chain: wolf 1 follows the hero, wolf 2 follows wolf 1, and the hero walks 8 north.
console @Follow(GetObjectByName("Mv1"),GetLeader())
console @Follow(GetObjectByName("Mv2"),GetObjectByName("Mv1"))
console @MoveToPoint(GetLeader(),GSGetVar(0,"hx"),Add(GSGetVar(0,"hy"),6))
# (The walking hero's environment remark, acks 006D88E0, comes at random: in some 160 updates here,
# not in every run; units, map_objects and ui_quick_items reach it too.)
wait 7000

# Around an obstacle: an ogre stands on the straight line from a new wolf to its target.
console @AddUnitToServer("MvRock","OgreTutor","MvRock",Add(GSGetVar(0,"hx"),4),Sub(GSGetVar(0,"hy"),4),0)
console @AddUnitToServer("Mv4","WolfTutor","Mv4",Add(GSGetVar(0,"hx"),1),Sub(GSGetVar(0,"hy"),4),0)
waitfor 10 IsAlive(GetObjectByName("MvRock"))
waitfor 10 IsAlive(GetObjectByName("Mv4"))
console @UMClear(GetObjectByName("MvRock"))
console @UMClear(GetObjectByName("Mv4"))
console @MoveToPoint(GetObjectByName("Mv4"),Add(GSGetVar(0,"hx"),7),Sub(GSGetVar(0,"hy"),4))
wait 2000
check IsLess(DistanceUnitPoint(GetObjectByName("MvRock"),Add(GSGetVar(0,"hx"),4),Sub(GSGetVar(0,"hy"),4)),2)
check IsAlive(GetLeader())
check IsAlive(GetObjectByName("Mv1"))
check IsAlive(GetObjectByName("Mv2"))
check IsAlive(GetObjectByName("Mv3"))
# Movement modes: Walk, Run and Crawl as standalone commands.
console @Walk(GetObjectByName("Mv1"))

console @Crawl(GetObjectByName("Mv3"))
wait 1000
check IsAlive(GetLeader())

# ---- section ui_quicksave
# The quick save's picture (R13; docs/notes/renderer.md section 6, "Save / load screen's pictures"):
# the battle screen's F5 (action 0x35) saves into "saveq" through 006570E0 with no frame drawn first (DL 0), so
# CaptureScreen (005FD3D0) copies the screen between frames, before the next frame's draws: the last
# frame, as the original's render target holds it (the image once cleared its frame there and saved
# plain blue). checksave compares the save's shot.sav (its 5:6:5 picture and the full-colour one after it)
# with a capture of the same frozen moment (in a window wider than 4:3 the copy is the 800x600 layout's
# centred 4:3 rectangle: W12, "centre"); then the Load window shows the pictures (saveq selected, then savea,
# whose shot.sav oldshot cuts to the 5:6:5 picture the original game writes: an old save's picture).
# Back to the main menu for the next section.

waitwindow 10 escmenu
clickregion 3
frames 5
key enter
waitwindow 15 menu

waitfor 15 IsGreater(HP(GetLeader()),0)

frames 5

key f5
