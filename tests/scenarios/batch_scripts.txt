# ---- section script_parse
# Script parser: console lines go through CScript::Load ("@x" loads "WorldScript(x)", "#x" loads
# "WorldScript( ConsoleFloat( x ))", "$x" loads "WorldScript( ConsoleString( x ))"). Covers nested
# calls of every argument type, literals and comments, global and local variables, assignments,
# script definitions and calls, and one deliberate error per parser message (checked with expect;
# the console prints "Script error: <message>").
#
# Names a console line declares are qualified with the number of files loaded so far ("!N!name"),
# and only a successful load counts a file. So definition lines end with a deliberate top-level error
# (EndMarker...), and the very next line uses the names. Identifiers avoid keyword prefixes (the
# lexer splits "Run1" into "Run" and "1").
waitfor 15 IsGreater(HP(GetLeader()),0)

# Nesting, literals (negative, leading '.', keywords in any case), strings, objects and groups.
check IsEqual(Mul(Add(1,2),Sub(10,4)),18)
check IsEqual(Div(Add(Mul(2,3),Sub(9,3)),Add(1,1)),6)
check IsEqual(Add(-1.5,.5),-1)
check isequal(ADD(1,2),3)
check IsEqualString("a b","a b")
check Not(IsEqualString("abc","abd"))
check IsEqual(GetObjectID(GetLeader()),GetObjectID(GetLeader()))
check GroupHas(GetPlayerUnits(GetPlayer(GetLeader())),GetLeader())

# ---- section save_flight
# A save and a quick load with effects in flight (H43c; src/game/effects/world_effect_missile.cpp EffectArrow::Load,
# world_effect.cpp EffectSoundSeq::Load, ui_saveload.cpp QuickLoad). The hero casts a fire wall (its sound sequence
# plays while the wall burns) and a magic arrow at a wolf 70 squares away; the arrow needs some six seconds to get
# there. F5 quick-saves into "saveq" while it is in the air, the wolf is killed on arrival, then F8 quick-loads: the
# wolf is alive again at its full health, the arrow and the wall's sound are read from the save, the arrow flies on
# and kills the wolf a second time.
# Back to the main menu for the next section.
frames 5
ifwindow battle key esc
waitwindow 10 escmenu
clickregion 3
frames 5
key enter
waitwindow 15 menu
newgame
waitline 15 Saved savea
waitfor 15 IsGreater(HP(GetLeader()),0)
console @BlockUnit(GetLeader(),0)
console @GodMode(0,1)
waitwindow 15 battle
console @GSSetVar(0,"sf.x",GetX(GetLeader())) GSSetVar(0,"sf.y",GetY(GetLeader()))
console @AddUnitToServer("FlyWolf","WolfTutor","FlyWolf",Add(GSGetVar(0,"sf.x"),70),GSGetVar(0,"sf.y"),0)
waitfor 10 IsAlive(GetObjectByName("FlyWolf"))
console @GSSetVar(0,"sf.hp",HP(GetObjectByName("FlyWolf")))
console @CastSpellPoint("firewall{d2;d2;e2}",Add(GSGetVar(0,"sf.x"),6),Add(GSGetVar(0,"sf.y"),9),Add(GSGetVar(0,"sf.x"),6),Add(GSGetVar(0,"sf.y"),13))
console @CastSpellUnit("arrow{e2}",GSGetVar(0,"sf.x"),GSGetVar(0,"sf.y"),GetObjectByName("FlyWolf"))
frames 6
check IsEqual(HP(GetObjectByName("FlyWolf")),GSGetVar(0,"sf.hp"))
key f5
waitline 15 Saved saveq
# The arrow arrives: the wolf dies.
waitfor 20 Not(IsAlive(GetObjectByName("FlyWolf")))

# F8: the quick save is loaded; the wolf stands again.
key f8
waitline 15 Loaded saveq
check IsAlive(GetObjectByName("FlyWolf"))
check IsEqual(HP(GetObjectByName("FlyWolf")),GSGetVar(0,"sf.hp"))
dump loaded_flight
waitfor 20 Not(IsAlive(GetObjectByName("FlyWolf")))
check IsAlive(GetLeader())

report
