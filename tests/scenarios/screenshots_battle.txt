# D7: release-notes screenshot of a fight with spells on zone1 (docs/screenshots/battle_spells.png, Rebirth's data):
# a wolf, two goblins, a skeleton and a boar attack the hero (GodMode), the camera zoomed in, a fire wall, fireballs and
# lightning cast by script while they close in.
# Not in the suites (run_all leaves screenshots_* out): run it by name, one game per work.py call (docs/notes/harness.md,
# "Screenshots"), and look at the frame before it replaces the PNG. Frames only, no golden.
resolution 3840x2160
widescreen 1
msaa 8
anisotropy 16
timeout 60
reject Script error
waitmenu
console thingamabob
newgame
waitfor 15 IsGreater(HP(GetLeader()),0)
waitwindow 15 battle
console @GodMode(0,1)
console @FixWorldTime(13)
climate 0
console filter all on
# Both side panels slid out (the map's sling button, the unit panel's: ui_widescreen_hud.txt).
click 793 15
frames 3
click 7 15
frames 3
console @GSSetVar(0,"hx",GetX(GetLeader()))
console @GSSetVar(0,"hy",GetY(GetLeader()))
# Each spawned creature takes the textures the maps place its kind with ("skin": AddUnitToServer names none).
console @AddUnitToServer("F1","WolfTutor","F1",Add(GSGetVar(0,"hx"),4),GSGetVar(0,"hy"),0)
skin F1 Wolf00 GoblinAxeBag00
console @AddUnitToServer("F2","GoblinF1","F2",Add(GSGetVar(0,"hx"),5),Add(GSGetVar(0,"hy"),2),0)
skin F2 Goblin00 GoblinAxeBag00
console @AddUnitToServer("F3","GoblinF1","F3",Add(GSGetVar(0,"hx"),4),Sub(GSGetVar(0,"hy"),2),0)
skin F3 Goblin00 GoblinAxeBag00
console @AddUnitToServer("F4","SkeletonF2","F4",Add(GSGetVar(0,"hx"),6),Add(GSGetVar(0,"hy"),1),0)
skin F4 Skeleton00 default0
console @AddUnitToServer("F5","BoarTutor","F5",Add(GSGetVar(0,"hx"),6),Sub(GSGetVar(0,"hy"),1),0)
skin F5 Boar00 default0
waitfor 15 IsAlive(GetObjectByName("F5"))
console @SetPlayer(GetObjectByName("F1"),6)
console @SetPlayer(GetObjectByName("F2"),6)
console @SetPlayer(GetObjectByName("F3"),6)
console @SetPlayer(GetObjectByName("F4"),6)
console @SetPlayer(GetObjectByName("F5"),6)
console @Attack(GetObjectByName("F1"),GetLeader())
console @Attack(GetObjectByName("F2"),GetLeader())
console @Attack(GetObjectByName("F3"),GetLeader())
console @Attack(GetObjectByName("F4"),GetLeader())
console @Attack(GetObjectByName("F5"),GetLeader())
# Two wheel turns zoom the camera in.
wheel 400 300 600
frames 3
wheel 400 300 600
frames 3
move 400 20
wait 800
console @CastSpellPoint("firewall{d2;e2}",GSGetVar(0,"hx"),GSGetVar(0,"hy"),Add(GSGetVar(0,"hx"),3),Add(GSGetVar(0,"hy"),2))
console @CastSpellUnit("fireball{e2}",GSGetVar(0,"hx"),GSGetVar(0,"hy"),GetObjectByName("F4"))
console @CastSpellUnit("lightning{e2;e2;e2;e2;e2}",GSGetVar(0,"hx"),GSGetVar(0,"hy"),GetObjectByName("F1"))
console @CastSpellPoint("inv_lit{ee3}",GSGetVar(0,"hx"),GSGetVar(0,"hy"),Add(GSGetVar(0,"hx"),5),Sub(GSGetVar(0,"hy"),2))
# The fireball's trail and the lightning's sparks are in the air half a second after the casts.
wait 250
freeze
capture battle_early
unfreeze
wait 250
freeze
capture battle
unfreeze
report
quit
