# Map objects (src/game/map_object.cpp, map_object_companion.cpp): the per-tick updates every map
# object runs (CMapObject::Update; CMapObjectServer::Update counting effects down and removing
# expired ones; CMapObjectClientSpecific::Update with effect end sounds and lights), scale effects
# (RecomputeScale, ResetScale), effects cleared on death (ClearEffects), the invisible flag
# (HideObject sets flags_ 0x10000, read by IsInvisible), and units being spawned, walked, turned,
# placed (SetCPFast), killed and removed. The verify totals in the report show what ran.
# SetCPFast(object, x, y, z); HideObject(object, on). No lever is known on zone1, so
# SwitchLeverState is not used. The game runs ~5.5x faster (10 ms per world update).
reject Script error
wait 6000
console thingamabob
newgame
speed 10
waitfor 15 IsGreater(HP(GetLeader()),0)
# The level intro keeps the hero blocked; unblock it instead of waiting.
console @BlockUnit(GetLeader(),0)

# Effects on the hero count down each world update; shrink rescales him.
console @CastSpellUnit("speed{d1}",GetX(GetLeader()),GetY(GetLeader()),GetLeader())
console @CastSpellUnit("shrink{e2;e2;d2;d1}",GetX(GetLeader()),GetY(GetLeader()),GetLeader())
wait 2000
check IsAlive(GetLeader())

# A wolf: spawned 5 east of the hero, walked back towards him, turned, put down elsewhere.
console @AddUnitToServer("Wolf1","WolfTutor","Wolf1",Add(GetX(GetLeader()),5),GetY(GetLeader()),0)
waitfor 10 IsAlive(GetObjectByName("Wolf1"))
console @GSSetVar(0,"wx",GetX(GetObjectByName("Wolf1")))
# The order is sometimes ignored right after the spawn (on the original too, 2 runs in 8): give the
# wolf a moment, and repeat the order once.
wait 1000
console @MoveToPoint(GetObjectByName("Wolf1"),Add(GetX(GetLeader()),2),GetY(GetLeader()))
wait 2000
console @MoveToPoint(GetObjectByName("Wolf1"),Add(GetX(GetLeader()),2),GetY(GetLeader()))
waitfor 15 IsLess(GetX(GetObjectByName("Wolf1")),Sub(GSGetVar(0,"wx"),2))
console @RotateTo(GetObjectByName("Wolf1"),GetX(GetLeader()),Add(GetY(GetLeader()),5))
wait 1000
console @SetCPFast(GetObjectByName("Wolf1"),Add(GetX(GetLeader()),4),Add(GetY(GetLeader()),2),GetZValue(Add(GetX(GetLeader()),4),Add(GetY(GetLeader()),2)))
wait 500
check IsLess(DistanceUnitUnit(GetLeader(),GetObjectByName("Wolf1")),6)

# Hidden and shown again.
console @HideObject(GetObjectByName("Wolf1"),1)
wait 1000
console @HideObject(GetObjectByName("Wolf1"),0)
wait 500

# An effect on the wolf, then its death clears its effects.
console @CastSpellUnit("speed{d1}",GetX(GetObjectByName("Wolf1")),GetY(GetObjectByName("Wolf1")),GetObjectByName("Wolf1"))
wait 1000
console @InflictDamage(GetObjectByName("Wolf1"),1000)
waitfor 10 Not(IsAlive(GetObjectByName("Wolf1")))

# A second wolf, removed from the server.
console @AddUnitToServer("Wolf2","WolfTutor","Wolf2",Sub(GetX(GetLeader()),4),GetY(GetLeader()),0)
waitfor 10 IsAlive(GetObjectByName("Wolf2"))
console @RemoveUnitFromServer(GetObjectByName("Wolf2"))
wait 1000
check IsAlive(GetLeader())
report
quit
