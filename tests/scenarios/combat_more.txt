# Combat, second part (src/game/combat_actions.cpp): the end of an attack order (FinishAction,
# DeliverAttack), bow missiles, item spell triggers, damage reactions and death (OnDamaged,
# HitDirection, PropagateHostility, WitnessMurder, SetDead: "Unit 12 damaged to death"), and
# healing (Heal: "Unit 12 was healed"). Combat log lines go to a console stream that is filtered
# out by default, so the filter is opened first. Each expect only looks at the output since the
# previous expect. The skill code (science / stealing, UseSkill) has no script command that
# reaches it; it runs only from the player's orders.
#
# The archer is the unit prototype "Human Gipath Brigand A1" from database.res: a level 1 bandit
# whose first weapon is "bone short bow.decayed bones", so its attacks go through the missile
# branch of DeliverAttack. WolfTutor dies to the first hit (as in combat.txt).
reject Script error
waitmenu
console thingamabob
console filter all on
newgame
waitfor 15 IsGreater(HP(GetLeader()),0)
# The level intro keeps the hero blocked; unblock it instead of waiting.
console @BlockUnit(GetLeader(),0)

# A death: the hero kills a wolf.
console @AddUnitToServer("Wolf1","WolfTutor","Wolf1",Add(GetX(GetLeader()),3),GetY(GetLeader()),0)
waitfor 10 IsAlive(GetObjectByName("Wolf1"))
console @SetPlayer(GetObjectByName("Wolf1"),6)
check IsEnemy(GetObjectByName("Wolf1"),0)
console @Attack(GetLeader(),GetObjectByName("Wolf1"))
waitfor 20 Not(IsAlive(GetObjectByName("Wolf1")))
expect damaged to death
check IsAlive(GetLeader())

# A second enemy attacks the hero until the hero is hurt. (The planned archer, "Human Gipath
# Brigand A1", neither attacked nor died within 20 s on the original game either, so a ranged
# attack is not covered yet; DeliverAttack's melee path is.)
console @AddUnitToServer("Archer1","WolfGreyF1","Archer1",Add(GetX(GetLeader()),3),GetY(GetLeader()),0)
waitfor 10 IsAlive(GetObjectByName("Archer1"))
console @SetPlayer(GetObjectByName("Archer1"),6)
check IsEnemy(GetObjectByName("Archer1"),0)
console #HP(GetLeader())
console @GSSetVar(0,"hp0",HP(GetLeader()))
# An order given right after the spawn is sometimes lost (about 1 run in 30 under 16 parallel
# games, on the original too): the attacker never swings. Give the new unit a moment, then
# repeat the order once; a repeated order on the same target only renews it.
wait 1000
console @Attack(GetObjectByName("Archer1"),GetLeader())
wait 1000
console @Attack(GetObjectByName("Archer1"),GetLeader())
waitfor 20 IsLess(HP(GetLeader()),GSGetVar(0,"hp0"))
expect was attacked by
console #HP(GetLeader())

# Healing: the healing spell reaches CUnitServer slot 21 (Heal), which logs it.
console @CastSpellUnit("healing{e2;e2;e2;e2}",GetX(GetLeader()),GetY(GetLeader()),GetLeader())
waitexpect 10 was healed
console #HP(GetLeader())

# The hero fights back until the archer is dead.
console @Attack(GetLeader(),GetObjectByName("Archer1"))
waitfor 20 Not(IsAlive(GetObjectByName("Archer1")))
expect damaged to death
check IsAlive(GetLeader())

# An attack that ends before it lands (CUnitServer::EndOrder, 00549690: UpdateState finds the acting
# unit's target out of reach): the hero attacks wolves that are put down 14 units away a few world
# updates later (wait 55 is one update); the right moment differs with the wolf's turn and step, so
# four tries, the delays 3, 4, 6 and 8 updates.
console @GodMode(0,1)
console @GSSetVar(0,"hx",GetX(GetLeader()))
console @GSSetVar(0,"hy",GetY(GetLeader()))
console @AddUnitToServer("Far0","WolfTutor","Far0",Add(GSGetVar(0,"hx"),3),GSGetVar(0,"hy"),0)
waitfor 10 IsAlive(GetObjectByName("Far0"))
console @SetPlayer(GetObjectByName("Far0"),6)
console @UMClear(GetObjectByName("Far0"))
console @Attack(GetLeader(),GetObjectByName("Far0"))
wait 165
console @SetCPFast(GetObjectByName("Far0"),Add(GSGetVar(0,"hx"),14),Add(GSGetVar(0,"hy"),0),GetZValue(Add(GSGetVar(0,"hx"),14),Add(GSGetVar(0,"hy"),0)))
wait 300
console @AddUnitToServer("Far1","WolfTutor","Far1",Add(GSGetVar(0,"hx"),3),GSGetVar(0,"hy"),0)
waitfor 10 IsAlive(GetObjectByName("Far1"))
console @SetPlayer(GetObjectByName("Far1"),6)
console @UMClear(GetObjectByName("Far1"))
console @Attack(GetLeader(),GetObjectByName("Far1"))
wait 220
console @SetCPFast(GetObjectByName("Far1"),Add(GSGetVar(0,"hx"),14),Add(GSGetVar(0,"hy"),2),GetZValue(Add(GSGetVar(0,"hx"),14),Add(GSGetVar(0,"hy"),2)))
wait 300
console @AddUnitToServer("Far2","WolfTutor","Far2",Add(GSGetVar(0,"hx"),3),GSGetVar(0,"hy"),0)
waitfor 10 IsAlive(GetObjectByName("Far2"))
console @SetPlayer(GetObjectByName("Far2"),6)
console @UMClear(GetObjectByName("Far2"))
console @Attack(GetLeader(),GetObjectByName("Far2"))
wait 330
console @SetCPFast(GetObjectByName("Far2"),Add(GSGetVar(0,"hx"),14),Add(GSGetVar(0,"hy"),4),GetZValue(Add(GSGetVar(0,"hx"),14),Add(GSGetVar(0,"hy"),4)))
wait 300
console @AddUnitToServer("Far3","WolfTutor","Far3",Add(GSGetVar(0,"hx"),3),GSGetVar(0,"hy"),0)
