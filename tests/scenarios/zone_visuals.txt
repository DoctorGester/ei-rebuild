# R57c: the client's picture kept beside a zone left (saves\current\<zone>.visuals.sav beside <zone>.sav,
# src/game/client_visuals.h; docs/notes/renderer.md, "What a save keeps of the picture"). Zone1 by day in rain: two
# wolves bleed and the hero walks (blood and footprints), the ground and the hero wet. Out to the town bz1g (LeaveZone
# keeps zone1, the client its picture beside it) and back: zone1 comes back from saves\current with its prints (aged
# by the ticks the zone ran meanwhile), the ground and the hero as wet as they were left and the clouds' cover going on
# from the kept one. The zone comes back still raining (the weather kept by the load, `weatherstate`; the original came
# back clear). Without the side file zone1 would come back with no prints, the ground and the hero dry at
# once and the cover at a clear sky's.
render checkall
resolution 1440x1080
reject Script error
timeout 120
waitmenu
console thingamabob
newgame
waitfor 15 IsGreater(HP(GetLeader()),0)
waitfor 15 IsUnitBlocked(GetLeader())
console @BlockUnit(GetLeader(),0)
waitwindow 15 battle
console @FixWorldTime(12)
console @SetWeather("rain")
waitpresent 10
move 400 20

# The hero walks out and back past two wolves, one hit and one killed (as in ground_prints).
console @GSSetVar(0,"hx",GetX(GetLeader()))
console @GSSetVar(0,"hy",GetY(GetLeader()))
console @MoveToPoint(GetLeader(),Add(GSGetVar(0,"hx"),6),GSGetVar(0,"hy"))
waitfor 20 IsGreater(GetX(GetLeader()),Add(GSGetVar(0,"hx"),5))
console @AddUnitToServer("PrintWolf1","WolfTutor","PrintWolf1",Add(GSGetVar(0,"hx"),3),Add(GSGetVar(0,"hy"),2),0)
console @AddUnitToServer("PrintWolf2","WolfTutor","PrintWolf2",Add(GSGetVar(0,"hx"),3),GSGetVar(0,"hy"),0)
waitfor 10 IsAlive(GetObjectByName("PrintWolf2"))
console @InflictDamage(GetObjectByName("PrintWolf1"),5)
console @InflictDamage(GetObjectByName("PrintWolf2"),100000)
wait 2000
console @MoveToPoint(GetLeader(),GSGetVar(0,"hx"),GSGetVar(0,"hy"))
waitfor 20 IsLess(GetX(GetLeader()),Add(GSGetVar(0,"hx"),1))
frames 2
weathercheck 0.3 0.9 0 0
unitweather leader 0.3 0.9 0 0
cloudcheck 0.5 0.6 0.5 0.8
groundprints 1 99 4 999 0 0
weatherstate 1 64

# Out to the town and back.
zone bz1g 0
waitzone 20 bz1g
waitfor 15 IsGreater(HP(GetLeader()),0)
wait 2000
zone gz1g 1
waitzone 20 gz1g
waitfor 15 IsGreater(HP(GetLeader()),0)
frames 2
weathercheck 0.6 0.95 0 0
unitweather leader 0.6 0.95 0 0
cloudcheck 0.5 0.6 0.6 0.8
groundprints 1 99 4 999 0 0
weatherstate 1 64
report
quit
